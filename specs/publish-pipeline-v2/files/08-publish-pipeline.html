Convex mutations

createPublishRun

// packages/database/convex/publish.ts
export const createPublishRun = mutation({
  args: {
    postId: v.id("posts"),
    socialProviderIds: v.optional(v.array(v.id("socialProviders"))),
    // if omitted, uses all providers on post
  },
  handler: async (ctx, args) => {
    const runId = crypto.randomUUID();
    const post = await getPost(ctx, args.postId);
    const providerIds = args.socialProviderIds ?? post.socialProviderIds;

    for (const providerId of providerIds) {
      await ctx.db.insert("publish_jobs", {
        postId: args.postId,
        socialProviderId: providerId,
        runId,
        status: "QUEUED",
        attemptCount: 0,
        maxAttempts: 5,
        userId: post.userId,
        organizationId: post.organizationId,
        createdAt: now,
        updatedAt: now,
      });
    }

    await ctx.db.patch(args.postId, {
      publishStatus: "QUEUED",
      activeRunId: runId,
      updatedAt: now,
    });

    return { runId, jobIds: [...] };
  },
});

completeAttempt

export const completeAttempt = internalMutation({
  args: {
    publishJobId: v.id("publish_jobs"),
    attemptNumber: v.number(),
    status: v.union(v.literal("SUCCEEDED"), v.literal("FAILED")),
    platformPostId: v.optional(v.string()),
    platformPostUrl: v.optional(v.string()),
    errorCode: v.optional(v.string()),
    errorClass: v.optional(errorClassSchema),
    errorMessage: v.optional(v.string()),
    workerRequestId: v.optional(v.string()),
  },
  handler: async (ctx, args) => {
    const job = await ctx.db.get(args.publishJobId);

    // Idempotency: already terminal → no-op
    if (job.status === "PUBLISHED" || job.status === "DEAD_LETTER") {
      return { alreadyComplete: true };
    }

    // Insert attempt row (idempotent on jobId + attemptNumber)
    await insertAttemptIfNotExists(ctx, args);

    if (args.status === "SUCCEEDED") {
      await ctx.db.patch(job._id, {
        status: "PUBLISHED",
        platformPostId: args.platformPostId,
        platformPostUrl: args.platformPostUrl,
        postedAt: now,
        completedAt: now,
        updatedAt: now,
      });
      await runPublishSideEffects(ctx, job, args);  // once only
    } else if (args.errorClass === "PERMANENT") {
      await ctx.db.patch(job._id, {
        status: "FAILED",
        errorClass: "PERMANENT",
        errorCode: args.errorCode,
        errorMessage: args.errorMessage,
        completedAt: now,
        updatedAt: now,
      });
    } else {
      // TRANSIENT — stay PROCESSING, increment attemptCount
      const newCount = job.attemptCount + 1;
      if (newCount >= job.maxAttempts) {
        await ctx.db.patch(job._id, {
          status: "DEAD_LETTER",
          attemptCount: newCount,
          completedAt: now,
          updatedAt: now,
        });
      } else {
        await ctx.db.patch(job._id, {
          attemptCount: newCount,
          errorClass: "TRANSIENT",
          errorCode: args.errorCode,
          errorMessage: args.errorMessage,
          updatedAt: now,
        });
      }
    }

    // Always re-derive post publishStatus
    await deriveAndPatchPostPublishStatus(ctx, job.postId);
    return { success: true };
  },
});

Side effects (run once on PUBLISHED)

Extracted from current updatePostPublishStatus into runPublishSideEffects:

  1. Upsert platformPosts[] on parent post
  2. replacePostAggregate for dashboard stats
  3. stats.addPublishDateInternal (streak)
  4. automations.linkPublishedPost (Instagram)
  5. Schedule video cleanup (7 days)

Stuck-job sweeper (Phase 5)

Convex cron job runs every 15 minutes:

  • Query publish_jobs where status = PROCESSING and updatedAt < now - 30min
  • If attemptCount < maxAttempts: reset to QUEUED, re-enqueue SQS
  • If attemptCount ≥ maxAttempts: transition to DEAD_LETTER
  • Re-derive post publishStatus
