HMAC authentication

All /internal/publish/* routes require HMAC-SHA256 authentication (ADR-004).

// Request headers
X-Delulu-Signature: sha256=<hex>
X-Delulu-Timestamp: <unix epoch seconds>

// Signature computation
const payload = `${timestamp}.${rawBody}`;
const signature = HMAC_SHA256(WORKER_HMAC_SECRET, payload);

// Validation
// 1. Reject if |now - timestamp| > 300 seconds (replay protection)
// 2. Constant-time compare signature
// 3. Reject if WORKER_HMAC_SECRET not configured

POST /internal/publish/start

Called by worker when SQS message is received. Transitions job QUEUED → PROCESSING and records attempt start.

// Request
{
  "publishJobId": "abc123",
  "attemptNumber": 1,
  "workerRequestId": "sqs-msg-uuid"
}

// Response 200
{
  "job": {
    "status": "PROCESSING",
    "socialType": "INSTAGRAM",
    "socialPublishInput": { ... }
  }
}

// Response 409 — job already terminal (idempotent)
{
  "alreadyComplete": true,
  "status": "PUBLISHED"
}

POST /internal/publish/complete

Called by worker after provider API call. Idempotent completion gateway.

// Request — success
{
  "publishJobId": "abc123",
  "attemptNumber": 1,
  "status": "SUCCEEDED",
  "platformPostId": "17895695668004550",
  "platformPostUrl": "https://instagram.com/p/...",
  "workerRequestId": "sqs-msg-uuid"
}

// Request — failure
{
  "publishJobId": "abc123",
  "attemptNumber": 2,
  "status": "FAILED",
  "errorCode": "RATE_LIMIT_EXCEEDED",
  "errorClass": "TRANSIENT",
  "errorMessage": "Rate limit exceeded for Instagram. Retry after 60s",
  "workerRequestId": "sqs-msg-uuid"
}

// Response 200
{
  "success": true,
  "jobStatus": "PROCESSING",       // or PUBLISHED, FAILED, DEAD_LETTER
  "postPublishStatus": "PROCESSING" // derived
}

// Response 409 — duplicate completion (idempotent no-op)
{
  "alreadyComplete": true
}

Implementation sketch

// apps/api/src/routes/internal/publish.ts

import { Hono } from "hono";
import { verifyHmac } from "@api/lib/hmac";
import { mapWorkerError } from "@api/lib/publish-errors";
import { convex } from "@delulu/database/node";

const publish = new Hono();

publish.use("*", verifyHmac);

publish.post("/complete", async (c) => {
  const body = await c.req.json();

  // Map worker error code → errorClass if not provided
  if (body.status === "FAILED" && !body.errorClass) {
    body.errorClass = mapWorkerError(body.errorCode);
  }

  const result = await convex.mutation(
    internal.publish.completeAttempt,
    body,
  );

  return c.json(result);
});

export { publish };

POST /internal/publish/enqueue

Called after createPublishRun to send SQS messages for each job.

// Request
{
  "runId": "uuid-v4",
  "jobs": [
    {
      "publishJobId": "abc123",
      "socialType": "INSTAGRAM",
      "socialPublishInput": { ... }
    }
  ]
}

// Response 200
{
  "enqueued": 3,
  "failed": 0
}

Consolidates queueing logic currently duplicated in post.service.ts and publishScheduledPost.

POST /v1/posts/:id/publish

User-facing publish endpoint (Clerk JWT). Replaces direct Convex mutation from dashboard.

// Request
POST /v1/posts/jx7abc/publish
Authorization: Bearer <clerk-jwt>

// Handler flow
1. Verify user owns post (or org member with publish permission)
2. Verify org approval if organizationId set
3. convex.mutation(createPublishRun, { postId })
4. POST /internal/publish/enqueue (or inline SQS send)
5. Return { runId, publishStatus: "QUEUED" }

// Response 200
{
  "runId": "uuid-v4",
  "publishStatus": "QUEUED",
  "jobCount": 3
}

POST /v1/posts/:id/retry

Retry failed platforms only. Creates new publish run for jobs in FAILED or DEAD_LETTER state.

// Request
POST /v1/posts/jx7abc/retry
Authorization: Bearer <clerk-jwt>
{
  "socialProviderIds": ["kh2..."]  // optional — default: all failed
}

// Handler flow
1. Query publish_jobs for post where status IN (FAILED, DEAD_LETTER)
2. Filter to requested socialProviderIds (or all failed)
3. createPublishRun with only those providerIds
4. Enqueue SQS messages
5. Return { runId, retriedCount }

// Response 200
{
  "runId": "uuid-v4",
  "retriedCount": 1,
  "publishStatus": "QUEUED"
}

Worker client change

// packages/worker/client.ts (v2)

export async function processMessage(messageBody: string) {
  const { publishJobId, attemptNumber, socialPublishInput, socialType } =
    JSON.parse(messageBody);

  // 1. Start attempt via API
  await apiPost("/internal/publish/start", {
    publishJobId, attemptNumber, workerRequestId: getRequestId(),
  });

  // 2. Execute provider
  const result = await providerRegistry[socialType].publish({ ... });

  // 3. Complete via API (never call Convex directly)
  if (result.isErr()) {
    const error = result.error;
    await apiPost("/internal/publish/complete", {
      publishJobId, attemptNumber,
      status: "FAILED",
      errorCode: error.code,
      errorClass: isRetryableError(error) ? "TRANSIENT" : "PERMANENT",
      errorMessage: error.message,
    });
    // Throw for TRANSIENT so SQS retries; return for PERMANENT
    if (isRetryableError(error)) throw error;
    return;
  }

  await apiPost("/internal/publish/complete", {
    publishJobId, attemptNumber,
    status: "SUCCEEDED",
    platformPostId: result.value.platformPostId,
    platformPostUrl: result.value.platformPostUrl,
  });
}
