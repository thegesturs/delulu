07 — Schema

Convex tables, indexes, and the single source of truth in schemas/publish.ts.

SSOT (ADR-009)

All publish enums and base schemas live in packages/database/convex/schemas/publish.ts. Worker error codes in packages/worker/providers/errors.ts map to errorClass at the API boundary — never redefined in Convex.

schemas/publish.ts (SSOT)

// packages/database/convex/schemas/publish.ts

import { v } from "convex/values";

// ── Enums ──────────────────────────────────────────────────

export const publishJobStatusSchema = v.union(
  v.literal("QUEUED"),
  v.literal("PROCESSING"),
  v.literal("PUBLISHED"),
  v.literal("FAILED"),
  v.literal("DEAD_LETTER"),
);

export const publishAttemptStatusSchema = v.union(
  v.literal("STARTED"),
  v.literal("SUCCEEDED"),
  v.literal("FAILED"),
);

export const publishStatusSchema = v.union(
  v.literal("QUEUED"),
  v.literal("PROCESSING"),
  v.literal("PARTIAL"),
  v.literal("PUBLISHED"),
  v.literal("FAILED"),
);

export const errorClassSchema = v.union(
  v.literal("TRANSIENT"),
  v.literal("PERMANENT"),
);

export const PUBLISH_JOB_STATUS = {
  QUEUED: "QUEUED",
  PROCESSING: "PROCESSING",
  PUBLISHED: "PUBLISHED",
  FAILED: "FAILED",
  DEAD_LETTER: "DEAD_LETTER",
} as const;

export const PUBLISH_STATUS = {
  QUEUED: "QUEUED",
  PROCESSING: "PROCESSING",
  PARTIAL: "PARTIAL",
  PUBLISHED: "PUBLISHED",
  FAILED: "FAILED",
} as const;

// ── publish_jobs ───────────────────────────────────────────

export const basePublishJobSchema = v.object({
  postId: v.id("posts"),
  socialProviderId: v.id("socialProviders"),
  runId: v.string(),                    // UUID grouping jobs in one publish run
  status: publishJobStatusSchema,
  attemptCount: v.number(),
  maxAttempts: v.number(),              // default 5
  nextRetryAt: v.optional(v.number()),
  errorClass: v.optional(errorClassSchema),
  errorCode: v.optional(v.string()),    // from worker errors.ts
  errorMessage: v.optional(v.string()),
  platformPostId: v.optional(v.string()),
  platformPostUrl: v.optional(v.string()),
  postedAt: v.optional(v.number()),
  userId: v.optional(v.id("users")),
  organizationId: v.optional(v.string()),
  createdAt: v.number(),
  updatedAt: v.number(),
  completedAt: v.optional(v.number()),
});

// ── publish_attempts ───────────────────────────────────────

export const basePublishAttemptSchema = v.object({
  publishJobId: v.id("publish_jobs"),
  attemptNumber: v.number(),
  status: publishAttemptStatusSchema,
  errorClass: v.optional(errorClassSchema),
  errorCode: v.optional(v.string()),
  errorMessage: v.optional(v.string()),
  workerRequestId: v.optional(v.string()),  // SQS messageId or Lambda requestId
  startedAt: v.number(),
  completedAt: v.optional(v.number()),
});

// ── posts extension ──────────────────────────────────────────

export const postPublishFieldsSchema = v.object({
  publishStatus: v.optional(publishStatusSchema),
  activeRunId: v.optional(v.string()),  // current publish run
});

schema.ts registration

// packages/database/convex/schema.ts (additions)

import {
  basePublishJobSchema,
  basePublishAttemptSchema,
} from "./schemas";

export default defineSchema({
  // ... existing tables ...

  publish_jobs: defineTable(basePublishJobSchema.fields)
    .index("by_post_id", ["postId"])
    .index("by_post_provider", ["postId", "socialProviderId"])
    .index("by_run_id", ["runId"])
    .index("by_status", ["status"])
    .index("by_status_next_retry", ["status", "nextRetryAt"])
    .index("by_user_id", ["userId"]),

  publish_attempts: defineTable(basePublishAttemptSchema.fields)
    .index("by_job_id", ["publishJobId"])
    .index("by_job_attempt", ["publishJobId", "attemptNumber"]),
});

posts table changes

Add to basePostSchema in schemas/posts_media.ts:

// Added fields on posts document
publishStatus: v.optional(publishStatusSchema),  // derived from jobs
activeRunId: v.optional(v.string()),             // current publish run UUID

posts.status retains lifecycle values (SAVED, SCHEDULED, DELETED). Values PUBLISHED, FAILED, PROCESSING migrate to publishStatus over Phase 1–3.

Error code mapping

Worker isRetryableError() maps to errorClass at the API layer:

  • NetworkError, RateLimitError, APIError (5xx) → TRANSIENT
  • InvalidAccessTokenError, NoContentError, InvalidMediaError → PERMANENT

Do not duplicate error class enums in Convex — import mapping function in apps/api.
