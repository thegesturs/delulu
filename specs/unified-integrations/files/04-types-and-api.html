Retryable classification

CoderetryableWhen
RATE_LIMITEDtrueHTTP 429
NETWORK_ERRORtrueTimeout, connection reset

INVALID_MEDIAfalseUser/content fix required

PlatformIntegration

export interface PlatformCapabilities {
  publish: boolean;
  analytics: boolean;
  supportsDM: boolean;
  multiStepConnect: boolean;
  supportsThreads: boolean;
  supportsStories: boolean;
  supportsDocuments: boolean;
}

export interface PlatformMeta {
  name: string;
  icon: string;
  editor: "normal" | "markdown" | "minimal";
  capabilities: PlatformCapabilities;
  toolTip?: string;
}

export interface PlatformAuth {
  scopes: string[];
  isMultiStep: boolean;
  getConnectUrl(ctx: ConnectContext): ConnectUrlResult | Promise<ConnectUrlResult>;
  handleCallback(input: CallbackInput): Promise<Response>;
  refreshToken?(input: { socialProviderId: string; refreshToken: string }): Promise<TokenRefreshResult>;
  onConnectStep?(input: ConnectStepInput): Promise<ConnectedAccount>;
}

export interface PlatformRules {
  maxLength: number | undefined;
  media: PlatformMediaRules;
  validate(input: ValidateInput): ValidationResult;
}

export interface PlatformIntegration {
  id: SocialType;
  meta: PlatformMeta;
  auth: PlatformAuth;
  rules: PlatformRules;
  settings: PlatformSettings;
  publish(ctx: PublishContext): Promise<PostResult>; // throws IntegrationError
  webhooks?: PlatformWebhooks;
  queries?: PlatformQueries;
}
