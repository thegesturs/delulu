Error model (replaces neverthrow)

// packages/integrations/src/errors.ts
export class IntegrationError extends Error {
  readonly code: string;
  readonly provider: string;
  readonly retryable: boolean;

  constructor(opts: {
    code: string;
    provider: string;
    message: string;
    retryable?: boolean;
  }) {
    super(opts.message);
    this.code = opts.code;
    this.provider = opts.provider;
    this.retryable = opts.retryable ?? false;
  }
}

// Provider contract
publish(ctx: PublishContext): Promise<PostResult>; // throws IntegrationError

// Worker boundary — packages/worker/client.ts
try {
  const result = await integration.publish(ctx);
  await convex.mutation(api.posts.updatePostPublishStatus, { status: "PUBLISHED", ... });
} catch (error) {
  const message =
    error instanceof IntegrationError ? error.message : "Unknown publish error";
  await convex.mutation(api.posts.updatePostPublishStatus, {
    status: "FAILED",
    failureReason: message,
    ...
  });
  if (error instanceof IntegrationError && error.retryable) {
    throw error; // allow SQS visibility-timeout retry
  }
}
